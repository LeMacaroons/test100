@echo off
setlocal EnableExtensions EnableDelayedExpansion
rem ====================================================================
rem  extract_call.cmd - extract one SIP call (SIP + RTP + RTCP) from
rem  large capture files using tshark, mergecap and editcap.
rem  Edit the values below. Leave a value empty to be prompted for it.
rem  Lists (ports, hosts) can be separated by spaces or commas.
rem ====================================================================
set "RAW_DIR=G:\nov18\sample1\sample1_raw"
set "OUT_DIR=G:\nov18\sample1\filtered"
set "CALLID=f6715507afdd255@50.2.2.11"
set "SIP_PORTS=5065"
set "MEDIA_PORTS=34704 33371 33634 33638"
set "HOSTS=60.1.15.200"
set "START=2026-11-18 16:22:00"
set "END=2026-11-18 17:00:00"
set "ADD_RTCP=1"
set "WS=C:\Program Files\Wireshark"
rem ====================================================================

if not defined RAW_DIR     set /p "RAW_DIR=Folder with raw captures: "
if not defined OUT_DIR     set /p "OUT_DIR=Output folder: "
if not defined CALLID      set /p "CALLID=SIP Call-ID: "
if not defined SIP_PORTS   set /p "SIP_PORTS=SIP port(s): "
if not defined MEDIA_PORTS set /p "MEDIA_PORTS=Media port(s): "
if not defined HOSTS       set /p "HOSTS=Host IP(s) to restrict to (Enter = none): "
if not defined START       set /p "START=Start (YYYY-MM-DD HH:MM:SS): "
if not defined END         set /p "END=End   (YYYY-MM-DD HH:MM:SS): "
for %%v in (RAW_DIR OUT_DIR CALLID SIP_PORTS MEDIA_PORTS START END) do (
  if not defined %%v (echo Missing value: %%v & exit /b 1)
)

set "TSHARK=%WS%\tshark.exe"
set "MERGECAP=%WS%\mergecap.exe"
set "EDITCAP=%WS%\editcap.exe"
set "CAPINFOS=%WS%\capinfos.exe"
if not exist "!TSHARK!"   (echo tshark.exe not found in "!WS!" - fix WS at the top & exit /b 1)
if not exist "!MERGECAP!" (echo mergecap.exe not found in "!WS!" - fix WS at the top & exit /b 1)

for /f "tokens=1 delims=@" %%a in ("!CALLID!") do set "CIDSHORT=%%a"
set "OUT=%OUT_DIR%\call_!CIDSHORT!.pcapng"
set "WORK=%OUT_DIR%\_work_!CIDSHORT!"
if not exist "%OUT_DIR%" mkdir "%OUT_DIR%"
if exist "%WORK%" rmdir /s /q "%WORK%"
mkdir "%WORK%"

rem ---- pass-1 filter: ports (+1 for RTCP) and optional hosts ----
rem      (uses eq/and/or/not instead of == && || ! so cmd leaves them alone)
set "PF="
for %%p in (%SIP_PORTS%) do set "PF=!PF! or tcp.port eq %%p or udp.port eq %%p"
for %%p in (%MEDIA_PORTS%) do (
  set "PF=!PF! or udp.port eq %%p"
  if "!ADD_RTCP!"=="1" (
    set /a R=%%p+1
    set "PF=!PF! or udp.port eq !R!"
  )
)
set "F1=(!PF:~4!)"
if defined HOSTS (
  set "HF="
  for %%h in (%HOSTS%) do set "HF=!HF! or ip.addr eq %%h"
  set "F1=!F1! and (!HF:~4!)"
)

rem ---- decode the custom SIP ports as SIP ----
set "DECODE="
for %%p in (%SIP_PORTS%) do set "DECODE=!DECODE! -d tcp.port==%%p,sip -d udp.port==%%p,sip"

echo.
echo Pass 1 filter: !F1!
echo.

rem ---- pass 1: filter each raw file ----
set /a N=0
for %%f in ("%RAW_DIR%\*") do (
  set "OK="
  for %%e in (.pcap .pcapng .cap) do if /i "%%~xf"=="%%e" set "OK=1"
  if defined OK (
    set /a N+=1
    echo [!N!] %%~nxf  ^(%%~zf bytes^)
    "!TSHARK!" -n -r "%%~ff" -o tcp.desegment_tcp_streams:FALSE -Y "!F1!" -w "!WORK!\part_!N!.pcapng" 2>nul
    if errorlevel 1 echo      WARNING: tshark reported an error ^(often a truncated last packet^)
  )
)
if "!N!"=="0" (echo No capture files found in "!RAW_DIR!" & exit /b 1)

rem ---- merge the pieces ----
echo.
echo Merging...
set "MERGED=%WORK%\merged.pcapng"
set "FIRST=1"
for %%p in ("%WORK%\part_*.pcapng") do (
  if defined FIRST (
    copy /y "%%~fp" "!MERGED!" >nul
    set "FIRST="
  ) else (
    "!MERGECAP!" -w "!WORK!\tmp.pcapng" "!MERGED!" "%%~fp"
    move /y "!WORK!\tmp.pcapng" "!MERGED!" >nul
  )
)

rem ---- cut to the time window ----
set "WIN=%WORK%\window.pcapng"
if exist "!EDITCAP!" (
  echo Trimming to !START! - !END!
  "!EDITCAP!" -A "!START!" -B "!END!" "!MERGED!" "!WIN!"
) else (
  echo editcap.exe not found - skipping the time trim
  copy /y "!MERGED!" "!WIN!" >nul
)

rem ---- pass 2: keep only this call's SIP (and the whole TCP streams it used) ----
set "CIDF=sip.Call-ID eq \"!CALLID!\""
"!TSHARK!" -n -r "!WIN!" !DECODE! -Y "!CIDF!" -T fields -e frame.number -e tcp.stream > "%WORK%\sip.txt" 2>nul
for %%A in ("%WORK%\sip.txt") do set "SIPSIZE=%%~zA"
if "!SIPSIZE!"=="0" (
  echo WARNING: no SIP found for this Call-ID ^(wrong ID or ports, or TLS^). Keeping all SIP.
  copy /y "!WIN!" "!OUT!" >nul
) else (
  set "F2=!CIDF! or (udp and not sip)"
  set "SEEN= "
  for /f "usebackq tokens=1,2" %%a in ("%WORK%\sip.txt") do (
    if not "%%b"=="" (
      if "!SEEN: %%b =!"=="!SEEN!" (
        set "SEEN=!SEEN!%%b "
        set "F2=!F2! or tcp.stream eq %%b"
      )
    )
  )
  "!TSHARK!" -n -r "!WIN!" !DECODE! -Y "!F2!" -w "!OUT!" 2>nul
)

rem ---- summary ----
echo.
echo Written: !OUT!
if exist "!CAPINFOS!" "!CAPINFOS!" -c -a -e -s "!OUT!"
(
  echo SIP messages for this Call-ID:
  "!TSHARK!" -n -r "!OUT!" !DECODE! -Y "!CIDF!" -T fields -E separator=/t -e frame.time -e ip.src -e ip.dst -e sip.Method -e sip.Status-Code -e sip.CSeq
  echo.
  echo RTP streams:
  "!TSHARK!" -n -r "!OUT!" --enable-heuristic rtp_udp -q -z rtp,streams
) > "!OUT!.summary.txt" 2>nul
type "!OUT!.summary.txt"

rmdir /s /q "%WORK%"
echo.
echo Open in Wireshark with filter:  sip.Call-ID eq "!CALLID!" or rtp or rtcp
endlocal
